composer create-project laravel/laravel:^12.0 ecommerce-core : tải dự án phiên bán 12

php artisan serve : chạy ứng dụng

php artisan make:migration create_categories_table : tạo file migration mới


php artisan migrate : chạy các file migration thì nó chạy method up trong từng file mà chưa chạy

php artisan migrate:rollback : chạy method down trong từng file chưa chạy

php artisan migrate:status : trạng thái các file migration

php artisan make:model Category : tạo entity cho bảng category


php artisan make:factory CategoryFactory : tạo factory 


php artisan db:seed : napj du lieu


php artisan install:api.php  tải file api.php ở routes

php artisan route:list --path=api  : xem các api ở api.php

php artisan make:controller Api/ProductController : tạo controller

php artisan make:request Api/StoreProductRequest : tạo request

php artisan make:policy ProductPolicy --model=Product : tạo policy cho phân quyền nâng cao
